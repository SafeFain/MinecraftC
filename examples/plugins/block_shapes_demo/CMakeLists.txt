cmake_minimum_required(VERSION 3.28)
project(block_shapes_demo LANGUAGES CXX)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
add_library(block_shapes_demo SHARED plugin.cpp)
target_include_directories(block_shapes_demo PRIVATE ../../../sdk)

set_target_properties(block_shapes_demo PROPERTIES PREFIX "")
if(APPLE)
    set_target_properties(block_shapes_demo PROPERTIES SUFFIX ".dylib")
endif()
