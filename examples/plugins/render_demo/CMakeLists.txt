cmake_minimum_required(VERSION 3.16)
project(MinecraftCExtensionDemo LANGUAGES CXX)
set(MINECRAFTC_SDK "${CMAKE_CURRENT_LIST_DIR}/../../../sdk" CACHE PATH "MinecraftC SDK directory")
add_library(render_demo MODULE plugin.cpp)
target_include_directories(render_demo PRIVATE "${MINECRAFTC_SDK}")
target_compile_features(render_demo PRIVATE cxx_std_17)
set_target_properties(render_demo PROPERTIES PREFIX "")
if(APPLE)
    set_target_properties(render_demo PROPERTIES SUFFIX ".dylib")
endif()
